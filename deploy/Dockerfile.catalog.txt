FROM mcr.microsoft.com/dotnet/aspnet:10.0 AS base
WORKDIR /app
EXPOSE 80
FROM mcr.microsoft.com/dotnet/sdk:10.0 AS build
WORKDIR /src
COPY . .
RUN dotnet build src\SagaSales.Catalog\SagaSales.Catalog.csproj -c Release -o /app/bin

FROM base AS publish
WORKDIR /app
COPY --from=build /app/bin ./bin

FROM base AS final
WORKDIR /app
COPY --from=publish /app ./app
ENTRYPOINT ["dotnet", "app/bin/SagaSales.Catalog.dll"]
